def unpack_gold(item):
    expected = item['gold'].get('expected_output', {})
    if item['gold']['expected_behavior'] == 'reject':
        return copy.deepcopy(item)
    if {'steps', 'final_output'}.issubset(expected):
        if not isinstance(expected['steps'], list) or not isinstance(expected['final_output'], dict):
            raise ValueError('Invalid archived Gold envelope')
        output = expected['final_output']
        prefix = 'final_output.'
    else:
        output = expected
        prefix = ''
    result = copy.deepcopy(item)
    result['gold']['expected_output'] = copy.deepcopy(output)
    tolerances = item['gold'].get('tolerance', {})
    if isinstance(tolerances, dict):
        result['gold']['tolerance'] = {
            key: {'__leaf_tolerances__': {p[len(prefix + key):].lstrip('.'): v
                   for p, v in tolerances.items() if p == prefix + key or p.startswith(prefix + key + '.')},
                  '__default__': tolerances.get(key, 1e-6)}
            for key in output
        }
    return result

def path_matches(actual, expected, tolerance=1e-6):
    if not isinstance(tolerance, dict) or '__leaf_tolerances__' not in tolerance:
        return original_matches(actual, expected, tolerance)
    table = tolerance['__leaf_tolerances__']
    default = tolerance.get('__default__', 1e-6)
    def visit(a, e, path):
        if isinstance(e, dict):
            return isinstance(a, dict) and all(k in a and visit(a[k],v,path+'.'+k if path else k) for k,v in e.items())
        if isinstance(e, list):
            return isinstance(a,list) and len(a)==len(e) and all(visit(x,y,path+'.'+str(i) if path else str(i)) for i,(x,y) in enumerate(zip(a,e)))
        return original_matches(a,e,table.get(path,default))
    return visit(actual,expected,'')

def call_identity(item, events, objects):
    inp = item.get('input', {})
    # A directly supplied sequence is itself the authorized scientific object.
    # Its auxiliary FASTA fixture is not a second record-selection request.
    # Keep duplicate-record/chain cases unresolved exactly as before.
    direct = (inp.get('input_type') == 'amino_acid_sequence'
              and isinstance(inp.get('sequence'), str)
              and not any(inp.get(k) is not None for k in ('record_id','record_ids','chain_id')))
    if direct:
        explicit = [o for o in objects if o['source'] == 'request_sequence']
        seen_hashes = {o['sequence_sha256'] for o in explicit}
        identity_objects = explicit + [o for o in objects if o['source'] != 'request_sequence'
                                      and o['sequence_sha256'] not in seen_hashes]
        # Removing the redundant fixture alias only affects sequence-value
        # candidate resolution. File-boundary checks still use all objects.
        resolved = original_call_identity(item, events, identity_objects)
        original = original_call_identity(item, events, objects)
        for r, old in zip(resolved, original):
            args=r['event'].get('resolved_arguments') or {}
            if not isinstance(args.get('sequence'),str) or sequence_hash(args['sequence']) not in seen_hashes:
                r.update({k:old[k] for k in ('identity_correct','reasons')})
            elif args.get('file_path') is not None:
                r.update({k:old[k] for k in ('identity_correct','reasons')})
        return resolved
    return original_call_identity(item, events, objects)

private.call_identity = call_identity

def score_run(item, row, events, files, review=None):
    normalized = unpack_gold(item)
    normalized_evidence = evidence_digest(normalized,row,events)
    raw_evidence = evidence_digest(item,row,events)
    adjusted = None
    if review:
        if review.get('evidence_sha256') != raw_evidence:
            raise ValueError('Stale raw-archive review evidence')
        adjusted = dict(review,evidence_sha256=normalized_evidence)
    score = private.score_run(normalized,row,events,files,adjusted)
    score['evidence_sha256'] = raw_evidence
    score['gold_envelope_unpacked'] = normalized['gold']['expected_output'] != item['gold'].get('expected_output',{})
    return score
